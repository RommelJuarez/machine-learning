def repeated_element(array):
    numeros_vistos=set()
    for numero in array:
        if numero in numeros_vistos:
            print(numero)
            return numero
        numeros_vistos.add(numero)

def first_last_uppercase(lenguajke):
    
    return pd.Series([lang[0].upper() + lang[1:-1] + lang[-1].upper() for lang in lenguajke])

# Your solution HERE !!!!
def fibonacci_generator():
    a, b = 0, 1
    while True:
        yield a
        a, b = b, a + b
